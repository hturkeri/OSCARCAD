# Drawing import tests — P0 bootstrap. FROZEN: work packages never edit this file (added from src/drawing).
#
#   tests/drawing/*.cpp       -> ONE gtest executable pc_drawing_tests; ctest names "drawing.<Suite>.<Test>"
#                                (test_*.cpp per WP + helpers such as GroundTruth support code)
#   tests/drawing/gui/*.cpp   -> one Qt executable per file, ctest "drawing.gui.<name>" (QPA offscreen)  [GUI]
#   tools/drawing_corpus      -> ctest fixture "drawing.corpus.generate" (runs before the tests when it exists)
#
# Compile definitions available to the tests (string paths, no trailing slash):
#   PC_DRAWING_TEST_DIR        tests/drawing (source)        fixtures: tests/drawing/fixtures/<wp>/
#   PC_DRAWING_CORPUS_DIR      tests/drawing/corpus          (small committed core set)
#   PC_DRAWING_CORPUS_GEN_DIR  <build>/drawing_corpus        (generated by pc_drawing_corpus before ctest)
#   PC_DRAWING_TESSDATA_DIR    share/persistcad/tessdata if it has eng.traineddata, else system tessdata, else ""
#   PC_DRAWING_TMP_DIR         <build>/tests/drawing/tmp     (scratch output)
#   PC_SAMPLES_DIR, PC_DOCS_DIR
#   + PC_DRAWING_HAVE_* feature flags (from pc_drawing)
include(GoogleTest)
set(PC_DRWT_DIR ${CMAKE_CURRENT_SOURCE_DIR})
set(PC_DRAWING_CORPUS_GEN_DIR ${CMAKE_BINARY_DIR}/drawing_corpus)
set(PC_DRAWING_TMP_DIR ${CMAKE_CURRENT_BINARY_DIR}/tmp)
file(MAKE_DIRECTORY ${PC_DRAWING_TMP_DIR})

set(PC_DRAWING_TESSDATA_DIR "")
if(EXISTS ${CMAKE_SOURCE_DIR}/share/persistcad/tessdata/eng.traineddata)
  set(PC_DRAWING_TESSDATA_DIR ${CMAKE_SOURCE_DIR}/share/persistcad/tessdata)
else()
  find_path(PC_SYSTEM_TESSDATA eng.traineddata
    PATHS /usr/share/tesseract-ocr/5/tessdata /usr/share/tesseract-ocr/4.00/tessdata /usr/share/tessdata
          /usr/local/share/tessdata "$ENV{MINGW_PREFIX}/share/tessdata" "$ENV{TESSDATA_PREFIX}"
    NO_DEFAULT_PATH)
  if(PC_SYSTEM_TESSDATA)
    set(PC_DRAWING_TESSDATA_DIR ${PC_SYSTEM_TESSDATA})
  endif()
endif()

set(PC_DRWT_DEFS
  PC_DRAWING_TEST_DIR="${PC_DRWT_DIR}"
  PC_DRAWING_CORPUS_DIR="${PC_DRWT_DIR}/corpus"
  PC_DRAWING_CORPUS_GEN_DIR="${PC_DRAWING_CORPUS_GEN_DIR}"
  PC_DRAWING_TESSDATA_DIR="${PC_DRAWING_TESSDATA_DIR}"
  PC_DRAWING_TMP_DIR="${PC_DRAWING_TMP_DIR}"
  PC_SAMPLES_DIR="${CMAKE_SOURCE_DIR}/samples"
  PC_DOCS_DIR="${CMAKE_SOURCE_DIR}/docs")

file(GLOB PC_DRAWING_TEST_SOURCES CONFIGURE_DEPENDS ${PC_DRWT_DIR}/*.cpp ${PC_DRWT_DIR}/*.h)
add_executable(pc_drawing_tests ${PC_DRAWING_TEST_SOURCES})
target_include_directories(pc_drawing_tests PRIVATE ${PC_DRWT_DIR})
target_link_libraries(pc_drawing_tests PRIVATE GTest::gtest_main pc_drawing pc_drawing_optdeps)
target_compile_definitions(pc_drawing_tests PRIVATE ${PC_DRWT_DEFS})
gtest_discover_tests(pc_drawing_tests
  TEST_PREFIX "drawing."
  WORKING_DIRECTORY ${PC_DRWT_DIR}
  DISCOVERY_TIMEOUT 120
  PROPERTIES LABELS drawing FIXTURES_REQUIRED drawing_corpus
             ENVIRONMENT "QT_QPA_PLATFORM=offscreen")

if(TARGET pc_drawing_corpus)
  # Contract for WP-1's generator: pc_drawing_corpus --out <dir> --samples <dir> --quick ; exit 0 on success.
  add_test(NAME drawing.corpus.generate
           COMMAND pc_drawing_corpus --out ${PC_DRAWING_CORPUS_GEN_DIR} --samples ${CMAKE_SOURCE_DIR}/samples --quick)
  set_tests_properties(drawing.corpus.generate PROPERTIES FIXTURES_SETUP drawing_corpus LABELS drawing
                       ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 900)
endif()

if(PC_BUILD_GUI AND TARGET pc_drawing_ui)
  find_package(Qt6 QUIET COMPONENTS Test)
  file(GLOB PC_DRAWING_GUI_SOURCES CONFIGURE_DEPENDS ${PC_DRWT_DIR}/gui/*.cpp)
  foreach(src ${PC_DRAWING_GUI_SOURCES})
    get_filename_component(name ${src} NAME_WE)
    add_executable(pc_drawing_gui_${name} ${src})
    target_include_directories(pc_drawing_gui_${name} PRIVATE ${PC_DRWT_DIR}/gui)
    target_link_libraries(pc_drawing_gui_${name} PRIVATE pc_drawing_ui Qt6::Widgets)
    if(TARGET Qt6::Test)
      target_link_libraries(pc_drawing_gui_${name} PRIVATE Qt6::Test)
    endif()
    target_compile_definitions(pc_drawing_gui_${name} PRIVATE ${PC_DRWT_DEFS})
    add_test(NAME drawing.gui.${name} COMMAND pc_drawing_gui_${name})
    set_tests_properties(drawing.gui.${name} PROPERTIES LABELS "drawing;drawing-gui" TIMEOUT 600
                         ENVIRONMENT "QT_QPA_PLATFORM=offscreen" FIXTURES_REQUIRED drawing_corpus)
  endforeach()
endif()
